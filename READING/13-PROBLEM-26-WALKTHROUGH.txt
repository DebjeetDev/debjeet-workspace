STEP 4 — CODE (Kal Khud Likhega. Aaj Sirf Dekh.)

    function removeDuplicates(nums) {

      // Khali array ho toh seedha 0
      if (nums.length === 0) {
        return 0;
      }

      // Pehla number toh hamesha rahega, isliye 1 se shuru
      let writePos = 1;

      // Padhne wali ungli: 1 se aakhir tak
      for (let readPos = 1; readPos < nums.length; readPos = readPos + 1) {

        // Agar pichle number se ALAG hai -> NAYA number mila
        if (nums[readPos] !== nums[readPos - 1]) {

          nums[writePos] = nums[readPos];   // likh do
          writePos = writePos + 1;          // ungli aage
        }
        // Same hai? Toh kuch mat karo. Chhod do.
      }

      return writePos;   // kitne alag numbers hain
    }
