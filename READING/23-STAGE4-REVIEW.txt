    /**
     * Chai-pani leak: Rs 100 se kam wale kharche ka TOTAL.
     * @param {Array<Array>} rows  [date, merchant, amount]
     * @returns {{status:number, ok:boolean, data?:number, error?:string}}
     */
    function getSmallSpendTotal(rows) {
      // Guard 1 — array hai bhi?
      if (!Array.isArray(rows)) {
        return { status: 400, ok: false, error: "Input must be a valid array!" };
      }

      // Guard 2 — khali ho to jhootha data mat de
      if (rows.length === 0) {
        return { status: 200, ok: true, data: 0 };
      }

      const total = rows.reduce((acc, cur) => {
        if (!Array.isArray(cur)) return acc;           // row hai bhi? (null se bachaav)
        const amount = cur[2];
        if (!Number.isFinite(amount)) return acc;      // number hai bhi?
        if (amount < 0) return acc;                    // refund = spend nahi
        return amount < 100 ? acc + amount : acc;      // ⭐ TYPE 2: fixed 100
      }, 0);

      return { status: 200, ok: true, data: total };
    }
